{%- comment -%} Renders a star rating (0..5, with half-star support). Usage: {% include stars.html rating=4.5 size="lg" %} Size: sm (default) | lg | xl {%- endcomment -%} {%- assign size_in = include.size -%} {%- if size_in == nil or size_in == "" -%}{%- assign size = 'sm' -%}{%- else -%}{%- assign size = size_in -%}{%- endif -%} {%- assign rating_raw = include.rating -%} {%- if rating_raw == nil or rating_raw == "" -%}{%- assign r = 0.0 -%}{%- else -%}{%- assign r = rating_raw | plus: 0.0 -%}{%- endif -%} {%- assign full = r | floor -%} {%- assign dec = r | minus: full -%} {%- assign has_half = false -%} {%- if dec >= 0.75 -%} {%- assign full = full | plus: 1 -%} {%- elsif dec >= 0.25 -%} {%- assign has_half = true -%} {%- endif -%} {%- assign half_pos = full | plus: 1 -%} {%- for i in (1..5) -%} {%- if i <= full -%} ★ {%- elsif has_half and i == half_pos -%} ★★ {%- else -%} ★ {%- endif -%} {%- endfor -%}
